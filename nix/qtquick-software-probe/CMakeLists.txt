cmake_minimum_required(VERSION 3.21)
project(qtquick-software-probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(qtquick-software-probe main.cpp)
target_compile_options(qtquick-software-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(qtquick-software-probe PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
install(TARGETS qtquick-software-probe RUNTIME DESTINATION bin)
install(FILES Probe.qml tile.png DESTINATION share/qtquick-software-probe)
