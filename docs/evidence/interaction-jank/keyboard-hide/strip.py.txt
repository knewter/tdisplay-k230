#!/usr/bin/env python3
"""Measure the light strip above the keyboard grip in grim captures.

Usage: strip.py CAPTURE.png...
The grip is a 56 px light band at the keyboard's top edge with a 96 px dark
line centred 24 px below its top. Above the grip, column x=1000 shows Foot's
dark background (the app) or the light card-coloured backdrop. Prints, per
capture, the grip top row and the count of contiguous light rows directly
above it: 0 means app pixels meet the keyboard.
"""
import sys
from PIL import Image
light = lambda p: sum(p[:3]) > 600
for f in sys.argv[1:]:
    im = Image.open(f).convert('RGB'); W, H = im.size; px = im.load()
    line = next((y for y in range(H) if not light(px[W // 2, y]) and light(px[W // 2 - 80, y])
                 and all(light(px[W // 2, y - k]) for k in range(1, 20) if y - k >= 0)), None)
    if line is None:
        print(f'{f}: no grip'); continue
    top = line - 24; n = 0
    while top - 1 - n >= 0 and light(px[1000, top - 1 - n]): n += 1
    print(f'{f}: grip_top={top} light_rows_above={n}')
