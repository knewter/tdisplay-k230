#!/usr/bin/env bash
# run.sh TAG SYSTEM [PULLS] -- injected shade pulls, then cadence.
TAG=$1; SYS=$2; N=${3:-10}
S=${SCRATCH:?scratch directory}
W=${WORKTREE:?repository checkout}
R="ssh -F ${SSH_CONFIG:?private ssh config} k230"
PY=/nix/store/v189xydz6qkcd4cbkixcmv91w8hbc560-python3-riscv64-unknown-linux-gnu-3.14.7/bin/python3
OUT=$S/shade/$TAG; mkdir -p $OUT || exit 1
$R "cat > /var/tmp/shade-pull.py" < $S/shade/shade-pull.py || exit 1
SW="sudo -u shell env SWAYSOCK=/run/shell/sway-ipc.sock /run/current-system/sw/bin/swaymsg"
$R "flock -n /tmp/k230-board.lock sh -c 'S0=@\$(date +%s); $SW card_shell benchmark injected >/dev/null; sudo $PY /var/tmp/shade-pull.py /var/tmp/shade-$TAG $SYS $N; $SW card_shell benchmark-stop >/dev/null; sudo sh -c \"journalctl -u shell --since \$S0 --no-pager -o short-unix > /var/tmp/shade-$TAG/shell-journal.txt\"'; rm -f /var/tmp/shade-pull.py" | tail -1
$R "sudo tar -C /var/tmp/shade-$TAG -cf - . && sudo rm -rf /var/tmp/shade-$TAG" | tar -C $OUT -xf -
python3 $W/docs/evidence/interaction-jank/drawer-profile/cadence.py.txt $OUT > $OUT/cadence.json
python3 -c "import json,sys;c=json.load(open('$OUT/cadence.json'));print('$TAG',c['distinct_presented_fps'],c['distinct_interval_ms'],c['callback_to_commit_ms'],c.get('sway_render_ms'),c.get('damage_px'))"
