#!/usr/bin/env python3
"""Own one temporary mode override; independent timed restoration."""
import sys,subprocess,time,socket
from pathlib import Path
p=Path('/run/systemd/system/shell.service.d/92-k230-paint-profile.conf')
contents='[Service]\nEnvironment=WLR_PIXMAN_DRAW_TRACE=1\n'
unit='k230-paint-profile-restore'
def call(args):subprocess.run(args,check=True,timeout=35,stdout=subprocess.DEVNULL)
mode=sys.argv[1]
if mode=='on':
 if p.exists():raise RuntimeError('mode override already occupied')
 p.parent.mkdir(parents=True,exist_ok=True);p.write_text(contents)
 call(['systemd-run','--quiet','--unit='+unit,'--on-active=180s',sys.executable,str(Path(__file__).resolve()),'watchdog'])
elif mode in ('off','watchdog'):
 if mode=='off':subprocess.run(['systemctl','stop',unit+'.timer'],check=False,timeout=20)
 if not p.exists():print('override already absent');raise SystemExit(0)
 if p.read_text()!=contents:raise RuntimeError('owned override changed; refusing removal')
 p.unlink()
else:raise SystemExit('invalid mode')
call(['systemctl','daemon-reload']);call(['systemctl','restart','shell.service','shell-ui.service'])
end=time.monotonic()+25
while True:
 try:
  with socket.socket(socket.AF_UNIX) as s:s.connect('/run/shell/sway-ipc.sock')
  break
 except OSError:
  if time.monotonic()>end:raise RuntimeError('shell IPC not restored')
  time.sleep(.2)
print('paint profile active' if mode=='on' else 'paint profile removed',flush=True)
