from pathlib import Path
import subprocess,datetime,json,shlex,time
b=Path(__file__).parent
ssh=['ssh','-F',str(Path.home()/'tmp/k230-coordination/ssh_config.private'),'k230']
tag=datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%SZ');remote='/tmp/k230-rotation-loop-variants-'+tag

subprocess.run(ssh+['mkdir -m 0700 '+shlex.quote(remote)],check=True,timeout=30)
for d in sorted(b.iterdir()):

  print(d.name,mode,p.returncode,flush=True);p.check_returncode()
