/* Exact untransformed XRGB8888/ARGB8888 -> RGB565 compositing for Pixman buffers.
 * Caller must have reset the source transform and use the supplied clip as
 * the destination's complete clip. Masks and other formats fall back. */
#ifndef K230_RGB565_BLIT_H
#define K230_RGB565_BLIT_H
#include <stdbool.h>
#include <stdint.h>
#include <string.h>
#include <pixman.h>
static inline uint16_t k230_rgb565_pixel(uint32_t p) {
    return (uint16_t)(((p >> 8) & 0xf800) | ((p >> 5) & 0x07e0) | ((p >> 3) & 0x001f));
}
/* Pixman's 8-bit premultiplied OVER, including saturated malformed inputs.
 * Expand RGB565 with replicated low bits, then use rounded division by255. */
static inline uint16_t k230_rgb565_over(uint32_t s, uint16_t d) {
    unsigned alpha = s >> 24;
    if (alpha == 255) return k230_rgb565_pixel(s);
    if (s == 0) return d;
    unsigned inv = 255-alpha;
    unsigned r = ((d>>8)&248) | (d>>13);
    unsigned g = ((d>>3)&252) | ((d>>9)&3);
    unsigned b = ((d<<3)&248) | ((d>>2)&7);
    r = r*inv + 128; r = ((r+(r>>8))>>8) + ((s>>16)&255);
    g = g*inv + 128; g = ((g+(g>>8))>>8) + ((s>>8)&255);
    b = b*inv + 128; b = ((b+(b>>8))>>8) + (s&255);
    if (r > 255) r = 255;
    if (g > 255) g = 255;
    if (b > 255) b = 255;
    return (uint16_t)(((r>>3)<<11) | ((g>>2)<<5) | (b>>3));
}
static inline bool k230_rgb565_blit(pixman_op_t op, pixman_image_t *source,
        pixman_image_t *mask, pixman_image_t *dest, int sx, int sy,
        int dx, int dy, int width, int height, const pixman_region32_t *clip) {
    if (!source || !dest || mask || (op != PIXMAN_OP_SRC && op != PIXMAN_OP_OVER) ||
            (pixman_image_get_format(source) != PIXMAN_x8r8g8b8 &&
             pixman_image_get_format(source) != PIXMAN_a8r8g8b8) ||
            pixman_image_get_format(dest) != PIXMAN_r5g6b5) return false;
    bool over_alpha = op == PIXMAN_OP_OVER && pixman_image_get_format(source) == PIXMAN_a8r8g8b8;
    int sw = pixman_image_get_width(source), sh = pixman_image_get_height(source);
    int dw = pixman_image_get_width(dest), dh = pixman_image_get_height(dest);
    int ss = pixman_image_get_stride(source), ds = pixman_image_get_stride(dest);
    uint8_t *src = (uint8_t *)pixman_image_get_data(source);
    uint8_t *dst = (uint8_t *)pixman_image_get_data(dest);
    if (!src || !dst || sw <= 0 || sh <= 0 || dw <= 0 || dh <= 0 ||
            sw > 16384 || sh > 16384 || dw > 16384 || dh > 16384 ||
            ss < sw * 4 || ds < dw * 2 || ss % 4 || ds % 2 ||
            (uintptr_t)src % 4 || (uintptr_t)dst % 2 ||
            width <= 0 || height <= 0 || sx < 0 || sy < 0 ||
            (int64_t)sx + width > sw || (int64_t)sy + height > sh ||
            (size_t)sh > SIZE_MAX / (size_t)ss ||
            (size_t)dh > SIZE_MAX / (size_t)ds) return false;
    size_t sb = (size_t)ss * sh, db = (size_t)ds * dh;
    uintptr_t sa = (uintptr_t)src, da = (uintptr_t)dst;
    if (sa <= da ? da - sa < sb : sa - da < db) return false;
    int64_t right = (int64_t)dx + width, bottom = (int64_t)dy + height;
    int left = dx > 0 ? dx : 0, top = dy > 0 ? dy : 0;
    if (right > dw) right = dw;
    if (bottom > dh) bottom = dh;
    if (right <= left || bottom <= top) return true;
    pixman_region32_t region;
    pixman_region32_init_rect(&region, left, top, (unsigned)(right-left), (unsigned)(bottom-top));
    if (clip) pixman_region32_intersect(&region, &region, (pixman_region32_t *)clip);
    int count;
    pixman_box32_t *boxes = pixman_region32_rectangles(&region, &count);
    for (int i = 0; i < count; i++) {
        pixman_box32_t b = boxes[i];
        for (int y = b.y1; y < b.y2; y++) {
            const uint32_t *s = (const uint32_t *)(src + (size_t)(sy+y-dy)*ss) + sx+b.x1-dx;
            uint16_t *d = (uint16_t *)(dst + (size_t)y*ds) + b.x1;
            int n = b.x2-b.x1;
            if (n && (uintptr_t)d % 4) { *d = over_alpha ? k230_rgb565_over(*s, *d) : k230_rgb565_pixel(*s); d++; s++; n--; }
            for (; n >= 2; n -= 2, s += 2, d += 2) {
                if (over_alpha && (s[0] & s[1] & UINT32_C(0xff000000)) != UINT32_C(0xff000000)) {
                    if (s[0]) d[0] = k230_rgb565_over(s[0], d[0]);
                    if (s[1]) d[1] = k230_rgb565_over(s[1], d[1]);
                    continue;
                }
                uint64_t p = (uint64_t)s[0] | (uint64_t)s[1] << 32;
                uint64_t v = ((p >> 8) & UINT64_C(0x0000f8000000f800)) |
                    ((p >> 5) & UINT64_C(0x000007e0000007e0)) |
                    ((p >> 3) & UINT64_C(0x0000001f0000001f));
                uint32_t packed = (uint32_t)v | (uint32_t)(v >> 16);
#if __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
                memcpy(__builtin_assume_aligned(d, 4), &packed, sizeof(packed));
#else
                d[0] = (uint16_t)packed; d[1] = (uint16_t)(packed >> 16);
#endif
            }
            if (n) *d = over_alpha ? k230_rgb565_over(*s, *d) : k230_rgb565_pixel(*s);
        }
    }
    pixman_region32_fini(&region);
    return true;
}
#endif
