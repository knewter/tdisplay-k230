import sys,time,subprocess
sys.path.insert(0,'/tmp')
from k230_demo import ROOT,console,screenshot
out=ROOT/'docs/evidence/splash-preserve-trial'
cam=subprocess.Popen(['python3','tools/capture-feature.py','keyboard-contrast-diagnostic','--output-dir',str(out),'--duration','38','--video-size','1920x1080','--rotate180','--provenance','unknown','--description','Temporary second keyboard with white keys and black labels under timeout, default keyboard hidden.
