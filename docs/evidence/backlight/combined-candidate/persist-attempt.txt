rollback() {
 status=$?
 trap - EXIT
 if [ "$status" -ne 0 ] && [ "$started" -eq 1 ]; then
  set +e
  restore_failed=0
  for name in Image initrd.uimg k230-tdisplay.dtb bootargs.txt; do
   cp "$previous/$name" "/boot/$name" || restore_failed=1
  done
  nix-env --profile /nix/var/nix/profiles/system --set "$profile_before" || restore_failed=1
  cd /boot || restore_failed=1
  sha256sum -c "$previous/SHA256SUMS" || restore_failed=1
  test "$(readlink -f /nix/var/nix/profiles/system)" = "$profile_before" || restore_failed=1
  sync || restore_failed=1
  if [ "$restore_failed" -eq 0 ]; then
   echo K230_PERSIST_ROLLBACK_VERIFIED
  else
   echo K230_PERSIST_ROLLBACK_INCOMPLETE
  fi
 fi
 exit "$status"
}
