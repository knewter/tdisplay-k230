#!/usr/bin/env python3
"""wakelog.py -- blocking (select) log of power-key, base-key and touch-down events
with the backlight value at that moment. Base keys are logged without their
key code. Read-only: devices are not grabbed. Output: /var/tmp/wakelog.log"""
import glob, os, select, struct, time
DEV = {"/dev/input/event0": "power-key", "/dev/input/event1": "touch", "/dev/input/event2": "base-key"}
EV = struct.Struct("llHHi")
BL = glob.glob("/sys/class/backlight/*/brightness")[0]
fds = {os.open(p, os.O_RDONLY): n for p, n in DEV.items()}
with open("/var/tmp/wakelog.log", "a", buffering=1) as out:
    out.write(f"{time.strftime('%FT%T%z')} start\n")
    while True:
        for fd in select.select(list(fds), [], [])[0]:
            data = os.read(fd, EV.size * 64)
            for i in range(0, len(data) - EV.size + 1, EV.size):
                _, _, typ, code, val = EV.unpack_from(data, i)
                name = fds[fd]
                hit = (typ == 1 and val == 1 and (name != "touch" or code == 330))
                if hit:
                    b = open(BL).read().strip()
                    detail = f" code={code}" if name == "power-key" else ""
                    out.write(f"{time.strftime('%T')}.{int(time.time()*1000)%1000:03d} {name} down{detail} brightness={b}\n")
